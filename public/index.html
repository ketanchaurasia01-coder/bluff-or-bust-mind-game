function render(s) { state = s; $("roomCode").textContent = s.code; $("round").textContent = `Round ${s.round} / ${s.totalRounds}`; $("phase").textContent = s.message || s.phase; const me = s.players.find(p => p.id === pid); $("wallet").textContent = `Tokens: ${me?.tokens ?? 0}`; $("score").textContent = `Points: ${me?.score ?? 0}`; $("players").innerHTML = s.players.map(p => `
${esc(p.name)} — ${p.score} pts, ${p.tokens} tokens ${p.id === s.hostId ? "(HOST)" : ""}
`).join(""); $("start").classList.toggle( "hidden", s.status !== "lobby" || s.hostId !== pid ); $("start").disabled = s.players.length < 2; $("play").classList.toggle( "hidden", s.status === "lobby" || !s.current ); if (!s.current) { $("options").innerHTML = ""; $("options").classList.add("hidden"); $("answerButtons").classList.add("hidden"); $("betControls").classList.add("hidden"); $("result").textContent = ""; $("next").classList.add("hidden"); return; } $("question").textContent = s.current.question; const claimant = s.players.find( p => p.id === s.current.claimantId ); $("claimant").textContent = "Featured player: " + (claimant?.name || "Player"); $("answerButtons").classList.add("hidden"); $("betControls").classList.add("hidden"); $("result").textContent = ""; $("next").classList.add("hidden"); const isMyAnswer = s.phase === "answer" && s.current.claimantId === pid; $("options").classList.toggle("hidden", !isMyAnswer); // Show answer options without rebuilding them on every refresh. if (isMyAnswer) { $("answerButtons").classList.remove("hidden"); if ($("options").dataset.q !== s.current.question) { $("options").dataset.q = s.current.question; $("options").innerHTML = ""; answer = ""; s.current.options.forEach(option => { const button = document.createElement("button"); button.className = "option"; button.textContent = option; button.onclick = () => { answer = option; document.querySelectorAll("#options .option") .forEach(item => item.classList.remove("selected")); button.classList.add("selected"); }; $("options").appendChild(button); }); } // Restore selected-answer styling after polling refreshes. document.querySelectorAll("#options .option").forEach(button => { button.classList.toggle( "selected", button.textContent === answer ); }); } // Show prediction controls to players who are allowed to predict. if ( s.phase === "bets" && s.current.claimantId !== pid ) { $("betControls").classList.remove("hidden"); } // Show the revealed answer and result. if (s.phase === "reveal") { const result = s.current.result || {}; $("result").textContent = `Correct answer: ${result.correctAnswer || "—"}. ` + `Submitted: ${result.submittedAnswer || "—"}. ` + ( result.bluff ? "The player bluffed." : "The player was honest." ); if (s.hostId === pid) { $("next").classList.remove("hidden"); } } else if (s.phase !== "answer" && s.phase !== "bets") { $("result").textContent = s.message || s.phase; } // Match-complete message. if (s.status === "finished") { $("result").textContent = "Match complete!"; $("next").classList.add("hidden"); } }
